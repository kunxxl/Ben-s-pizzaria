#staff dashboard
SELECT
rota.date,
staff.fist_name,
staff.last_name,
staff.hourly_rate,
shift.start_time,
shift.end_time,
((hour(timediff(shift.end_time,shift.start_time))*60)+(minute(timediff(shift.end_time,shift.start_time))))/60 AS hours_in_shift,
((hour(timediff(shift.end_time,shift.start_time))*60)+(minute(timediff(shift.end_time,shift.start_time))))/60 * hourly_rate AS staff_cost
FROM rota
LEFT JOIN staff ON rota.staff_id = staff.staff_id
LEFT JOIN shift ON rota.shift_id = shift.shift_id
